## 一、总则

### 1.1 设计目标
- 完全离线运行
- 支持加密（用户隐私）
- 高效查询（识别 / 路线 / 物种）
- 易于扩展
- 数据完整性

### 1.2 数据库选型

| 候选 | 选 | 原因 |
|---|---|---|
| **SQLite + sqlcipher** | ✅ | 嵌入式、加密、成熟、Qt 原生支持 |
| Realm | ❌ | 体积大 |
| LevelDB | ❌ | 不适合复杂查询 |
| Room（Android） | ❌ | 限制于 Android，不跨平台 |
| CoreData | ❌ | iOS only |

### 1.3 数据库文件

```
files/
├── routes.db          # 路线主库（明文）
├── species.db         # 物种主库（明文，可下载）
├── user.db            # 用户库（加密）
│   ├── 识别历史
│   ├── 收藏
│   ├── 应急清单
│   ├── 紧急联系人
│   └── 学习进度
├── firstaid.db        # 急救（明文）
├── survival.db        # 避险（明文）
├── knots.db           # 绳结（明文）
└── settings.db        # 设置（加密）
```

### 1.4 设计原则

| 原则 | 实施 |
|---|---|
| 命名规范 | snake_case，名词复数 |
| 主键 | TEXT（UUID）或 INTEGER AUTOINCREMENT |
| 时间 | TEXT（ISO 8601）或 INTEGER（Unix 时间戳）|
| 软删除 | is_deleted INTEGER DEFAULT 0 |
| 审计 | created_at, updated_at |
| 大字段 | 单独表或文件 |
| JSON | TEXT 字段存 JSON |

### 3.1 物种库（species.db）

#### 3.1.1 species - 物种主表

```sql
CREATE TABLE species (
    id TEXT PRIMARY KEY,
    scientific_name TEXT NOT NULL,           -- 拉丁学名
    common_name_cn TEXT NOT NULL,            -- 中文名
    alias TEXT,                              -- JSON: 别名列表
    category TEXT NOT NULL,                  -- plant/animal/snake/mushroom/insect
    family TEXT,                             -- 科
    genus TEXT,                              -- 属
    order_name TEXT,                         -- 目
    class_name TEXT,                         -- 纲
    
    -- 分布
    distribution_type TEXT,                  -- 全国/区域/省份/特定
    distribution_detail TEXT,                -- JSON: 详细分布
    min_lat REAL,
    max_lat REAL,
    min_lon REAL,
    max_lon REAL,
    min_elevation INTEGER,
    max_elevation INTEGER,
    habitats TEXT,                           -- JSON: 栖息地列表
    terrain_types TEXT,                      -- JSON: 地形
    
    -- 物候
    active_months TEXT,                      -- JSON: 活跃月份
    flowering_months TEXT,                   -- JSON: 花期
    fruiting_months TEXT,                    -- JSON: 果期
    hibernation_months TEXT,                 -- JSON: 冬眠期
    
    -- 风险
    danger_level INTEGER DEFAULT 0,          -- 0-5
    edible INTEGER DEFAULT 0,                -- 0/1
    edible_parts TEXT,                       -- JSON: 可食部位
    toxic_parts TEXT,                        -- JSON: 有毒部位
    
    -- 应对
    protective_gear TEXT,                    -- JSON: 防护装备
    first_aid_id TEXT,                       -- 关联急救条目
    countermeasures TEXT,                    -- JSON: 应对方法
    
    -- 描述
    description TEXT,
    features TEXT,                           -- JSON: 识别特征
    similar_species TEXT,                    -- JSON: 相似种
    
    -- 元数据
    commonality INTEGER DEFAULT 3,           -- 1-5 常见度
    iucn_status TEXT,                        -- LC/NT/VU/EN/CR
    protected_level TEXT,                    -- 保护级别
    source TEXT,                             -- 来源
    source_url TEXT,
    source_refs TEXT,                        -- JSON: 参考文献
    last_reviewed_at TEXT,
    reviewer TEXT,
    
    -- 审计
    created_at TEXT NOT NULL,
    updated_at TEXT NOT NULL,
    is_deleted INTEGER DEFAULT 0,
    
    CHECK (category IN ('plant', 'animal', 'snake', 'mushroom', 'insect')),
    CHECK (danger_level >= 0 AND danger_level <= 5)
);

-- 索引
CREATE INDEX idx_species_category ON species(category);
CREATE INDEX idx_species_danger ON species(danger_level);
CREATE INDEX idx_species_geo ON species(min_lat, max_lat, min_lon, max_lon);
CREATE INDEX idx_species_elevation ON species(min_elevation, max_elevation);
CREATE INDEX idx_species_updated ON species(updated_at);
```

## 六、数据迁移

### 6.1 版本管理

```cpp
class DatabaseMigrator {
public:
    struct Migration {
        int fromVersion;
        int toVersion;
        QString description;
        QStringList sql;       // 要执行的 SQL
        std::function<bool()> customFunc;  // 自定义迁移
    };
    
    bool migrate(Database db, int targetVersion) {
        int current = getCurrentVersion(db);
        
        if (current == targetVersion) return true;
        
        // 按顺序执行迁移
        for (const auto& m : migrations) {
            if (m.fromVersion == current) {
                // 执行
                executeInTransaction(db, m.sql);
                if (m.customFunc) m.customFunc();
                setCurrentVersion(db, m.toVersion);
                current = m.toVersion;
            }
        }
        
        return current == targetVersion;
    }
    
private:
    int getCurrentVersion(Database db) {
        QSqlQuery q(*connection(db));
        q.exec("PRAGMA user_version");
        if (q.next()) return q.value(0).toInt();
        return 0;
    }
    
    void setCurrentVersion(Database db, int version) {
        QSqlQuery q(*connection(db));
        q.exec(QString("PRAGMA user_version = %1").arg(version));
    }
};
```
