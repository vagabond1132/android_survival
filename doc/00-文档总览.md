> ```bash

> grep -rn 'Q_INVOKABLE' src --include=*.h --include=*.cpp | wc -l             # 353 —— 含注释里的提及
> grep -rn 'Q_INVOKABLE' src --include=*.h --include=*.cpp \
>   | grep -vE ':\s*(//|\*)' | wc -l                                          # 346 —— 去掉注释后的口径
> grep -c 'ctx->setContextProperty(' src/core/ServiceRegistry.cpp             # 19 —— 注入对象数
> grep -c 'setContextProperty'      src/core/ServiceRegistry.cpp             # 23 —— 含 4 处注释，别用这个数
> grep -cE 'std::make_unique' src/core/ServiceRegistry.cpp                    # 17 —— 服务对象数
> grep -cE '\*[[:space:]]*\w+\(\)[[:space:]]*const;' src/core/App.h           # 16 —— 声明式委托 getter；减掉 config() = 15 个服务 getter

> # 测试函数数: 必须只数 `private slots:` 段里的, 且排掉 49 个生命周期槽位
> awk 'FNR==1{sec=""} /^[[:space:]]*(private|protected|public)[[:space:]]+slots[[:space:]]*:/{sec="slots";next}
>      /^[[:space:]]*(private|protected|public)[[:space:]]*:/{sec=""}
